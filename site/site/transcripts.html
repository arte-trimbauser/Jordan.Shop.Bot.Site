function renderTranscripts(lista) { const container = document.getElementById('lista-transcripts'); if (!lista || lista.length === 0) { container.innerHTML = "

📭 Nenhum transcript encontrado.

"; return; } container.innerHTML = ''; lista.forEach((file, index) => { const nomeCompleto = file.name.replace('.html', ''); // Extrair o produto: ex: "ticket-shark_semanal-arte_10" → "shark_semanal" let produto = 'Produto'; const partes = nomeCompleto.split('-'); if (partes.length >= 2) { // Pode ser "ticket-shark_semanal-arte_10" → "shark_semanal" produto = partes.slice(1, partes.length - 1).join(' ') || 'Produto'; } const cliente = partes[partes.length - 1]?.toUpperCase() || 'CLIENTE'; const data = file.created_at ? new Date(file.created_at).toLocaleString('pt-PT') : 'Data desconhecida'; const id = `transcript-${index}`; const item = document.createElement('div'); item.className = 'transcript-item'; item.innerHTML = `
👤 ${cliente} — ${produto}
🕒 ${data}
🔗 ABRIR
`; container.appendChild(item); }); }
