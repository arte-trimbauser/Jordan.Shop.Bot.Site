function renderTranscripts(lista) { const container = document.getElementById('lista-transcripts'); if (!lista || lista.length === 0) { container.innerHTML = "

📭 Nenhum transcript encontrado.

"; return; } container.innerHTML = ''; lista.forEach((file, index) => { // Nome do ficheiro sem .html const nomeCompleto = file.name.replace('.html', ''); // Extrair produto e cliente do nome // Ex: "ticket-shark_semanal-arte_10" → produto="shark_semanal", cliente="arte_10" const partes = nomeCompleto.split('-'); let produto = 'Produto'; let cliente = 'CLIENTE'; if (partes.length >= 2) { // O produto são as partes do meio (ex: "shark_semanal") produto = partes.slice(1, partes.length - 1).join(' ') || 'Produto'; // O cliente é a última parte cliente = partes[partes.length - 1]?.toUpperCase() || 'CLIENTE'; } const data = file.created_at ? new Date(file.created_at).toLocaleString('pt-PT') : 'Data desconhecida'; const id = `transcript-${index}`; const item = document.createElement('div'); item.className = 'transcript-item'; item.innerHTML = `
👤 ${cliente} — ${produto}
🕒 ${data}
🔗 ABRIR
`; container.appendChild(item); }); }
